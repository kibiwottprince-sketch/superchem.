// ==================================
// SUPERCHEM: CHEMICAL FORMULA PARSER
// ==================================

function parseFormula(formula) {
    let position = 0;

    function parseGroup() {
        const atoms = {};

        while (position < formula.length) {
            const char = formula[position];

            if (char === ')' || char === ']') {
                break;
            }

            if (char === '(' || char === '[') {
                position++;

                const inner = parseGroup();

                if (position >= formula.length ||
                    (formula[position] !== ')' &&
                     formula[position] !== ']')) {
                    throw new Error("Unmatched bracket.");
                }

                position++;

                const multiplier = readNumber();

                for (const element in inner) {
                    atoms[element] = (atoms[element] || 0)
                        + inner[element] * multiplier;
                }

            } else {
                const match = formula.slice(position)
                    .match(/^([A-Z][a-z]?)/);

                if (!match) {
                    throw new Error("Invalid chemical formula.");
                }

                const element = match[1];
                position += element.length;

                const count = readNumber();

                atoms[element] = (atoms[element] || 0) + count;
            }
        }

        return atoms;
    }

    function readNumber() {
        const match = formula.slice(position).match(/^\d+/);

        if (!match) return 1;

        position += match[0].length;
        const number = Number(match[0]);

        if (number < 1) {
            throw new Error("Invalid atom multiplier.");
        }

        return number;
    }

    if (!formula || !/^[A-Za-z0-9()[\]]+$/.test(formula)) {
        throw new Error("Enter a valid chemical formula.");
    }

    const result = parseGroup();

    if (position !== formula.length) {
        throw new Error("Invalid brackets in formula.");
    }

    return result;
}

function checkAtoms() {
    const formula = document.getElementById("formula").value.trim();
    const output = document.getElementById("atomResult");

    try {
        const atoms = parseFormula(formula);

        let html = "<strong>Atom count for " + formula + "</strong><br>";

        let total = 0;

        for (const element in atoms) {
            html += element + ": " + atoms[element] + "<br>";
            total += atoms[element];
        }

        html += "<br><strong>Total atoms: " + total + "</strong>";

        output.innerHTML = html;

    } catch (error) {
        output.textContent = error.message;
    }
}


// ==================================
// EQUATION BALANCER
// ==================================

function gcd(a, b) {
    a = Math.abs(a);
    b = Math.abs(b);

    while (b) {
        const temp = b;
        b = a % b;
        a = temp;
    }

    return a || 1;
}

function balanceEquation() {
    const input = document.getElementById("equation").value.trim();
    const output = document.getElementById("balanceResult");

    try {
        const sides = input.split(/\s*(?:->|→|=)\s*/);

        if (sides.length !== 2) {
            throw new Error("Use an arrow between reactants and products.");
        }

        const reactants = sides[0].split("+").map(x => x.trim());
        const products = sides[1].split("+").map(x => x.trim());

        if ([...reactants, ...products].some(x => !x)) {
            throw new Error("One or more compounds are missing.");
        }

        const compounds = [...reactants, ...products];

        const compositions = compounds.map(parseFormula);

        const elements = [...new Set(
            compositions.flatMap(x => Object.keys(x))
        )].sort();

        const matrix = elements.map(element =>
            compositions.map((composition, index) => {
                const count = composition[element] || 0;
                return index < reactants.length ? count : -count;
            })
        );

        const rows = matrix.length;
        const cols = compounds.length;

        // Reduced row echelon form
        let pivotColumns = [];
        let pivotRow = 0;

        for (let col = 0; col < cols && pivotRow < rows; col++) {
            let selected = pivotRow;

            for (let r = pivotRow + 1; r < rows; r++) {
                if (Math.abs(matrix[r][col]) >
                    Math.abs(matrix[selected][col])) {
                    selected = r;
                }
            }

            if (Math.abs(matrix[selected][col]) < 1e-10) {
                continue;
            }

            [matrix[pivotRow], matrix[selected]] =
                [matrix[selected], matrix[pivotRow]];

            const divisor = matrix[pivotRow][col];

            for (let c = 0; c < cols; c++) {
                matrix[pivotRow][c] /= divisor;
            }

            for (let r = 0; r < rows; r++) {
                if (r === pivotRow) continue;

                const factor = matrix[r][col];

                for (let c = 0; c < cols; c++) {
                    matrix[r][c] -= factor * matrix[pivotRow][c];
                }
            }

            pivotColumns.push(col);
            pivotRow++;
        }

        const freeColumns = [];

        for (let c = 0; c < cols; c++) {
            if (!pivotColumns.includes(c)) {
                freeColumns.push(c);
            }
        }

        if (freeColumns.length !== 1) {
            throw new Error(
                "This equation needs a more advanced balancing method. " +
                "Try a simpler equation."
            );
        }

        const coefficients = Array(cols).fill(0);
        const freeColumn = freeColumns[0];

        coefficients[freeColumn] = 1;

        for (let r = pivotColumns.length - 1; r >= 0; r--) {
            const col = pivotColumns[r];
            coefficients[col] = -matrix[r][freeColumn];
        }

        // Convert decimal coefficients into integers.
        const scale = 1000000;

        let integers = coefficients.map(value =>
            Math.round(value * scale)
        );

        let divisor = integers.reduce((a, b) => gcd(a, b));

        integers = integers.map(value => value / divisor);

        if (integers.every(x => x < 0)) {
            integers = integers.map(x => -x);
        }

        if (integers.some(x => x <= 0)) {
            throw new Error("No positive balancing coefficients found.");
        }

        // Verify atom conservation.
        for (const element of elements) {
            let total = 0;

            compositions.forEach((composition, index) => {
                const sign = index < reactants.length ? 1 : -1;

                total += sign *
                    (composition[element] || 0) *
                    integers[index];
            });

            if (total !== 0) {
                throw new Error("Could not verify the balanced equation.");
            }
        }

        function formatCompound(compound, coefficient) {
            return (coefficient === 1 ? "" : coefficient) + compound;
        }

        const left = reactants.map((compound, index) =>
            formatCompound(compound, integers[index])
        );

        const right = products.map((compound, index) =>
            formatCompound(compound, integers[index + reactants.length])
        );

        const balanced = left.join(" + ") + " → " + right.join(" + ");

        output.innerHTML =
            "<strong>Balanced equation:</strong><br><br>" +
            "<span style='font-size:20px'>" + balanced + "</span><br><br>" +
            "<strong>Verification:</strong> All atoms are conserved.";

    } catch (error) {
        output.textContent = error.message;
    }
}

            name: "Combustion of propane",
            equation: "C3H8 + 5O2 → 3CO2 + 4H2O",

    const reaction = reactions[type];

        reaction.equation + "<br><br>" +
        reaction.explanation;
